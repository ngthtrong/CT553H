  \renewcommand*{\listtablename}{LIST OF TABLES}%

% ---- Covers ------------------------------------------------------------------
\newcommand{\ThesisFrame}{%
  \begin{tikzpicture}[remember picture,overlay]
    \draw[line width=1.6pt] ($(current page.north west)+(20mm,-20mm)$)
      rectangle ($(current page.south east)+(-20mm,20mm)$);
    \draw[line width=.5pt] ($(current page.north west)+(21.8mm,-21.8mm)$)
      rectangle ($(current page.south east)+(-21.8mm,21.8mm)$);
  \end{tikzpicture}}

  \thispagestyle{empty}\ThesisFrame\centering\setstretch{1.3}\fontsize{14bp}{18bp}\selectfont
}{\clearpage\restoregeometry}

% ---- Fixed front-matter pages -------------------------------------------------
% Unnumbered chapter heading that also appears in the table of contents; pages
% before the table of contents carry only the centred page number.
\newcommand{\FrontChapter}[1]{\addchap{#1}\thispagestyle{thesisplain}}
% Dotted writing lines for pages that are completed by hand.
\newcommand{\DottedLines}[1]{\foreach \i in {1,...,#1}{\par\noindent\dotfill\par}}

% ---- Numbering of the three parts -------------------------------------------
% PART 1 and PART 3 number their sections 1., 2., 3.1.; the chapters of PART 2
% (and the appendices) use 1.1., 1.1.1.
\newcommand{\ThesisFlatNumbering}{%
  \setcounter{section}{0}%
  \renewcommand*{\thesection}{\arabic{section}}%
  \renewcommand*{\thesubsection}{\thesection.\arabic{subsection}}%
  \renewcommand*{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}}
\newcommand{\ThesisChapterNumbering}{%
  \renewcommand*{\thesection}{\thechapter.\arabic{section}}%
  \renewcommand*{\thesubsection}{\thesection.\arabic{subsection}}%
  \renewcommand*{\thesubsubsection}{\thesubsection.\arabic{subsubsection}}}
